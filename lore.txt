<script>
const theorySelect = document.getElementById('theory-game');
Object.values(GAMES).forEach(g => {
  const o = document.createElement('option');
  o.value = g.id; o.textContent = g.title;
  theorySelect.appendChild(o);
});
document.getElementById('open-theory').addEventListener('click', () => openModal(document.getElementById('theory-modal')));
document.getElementById('theory-form').addEventListener('submit', e => {
  e.preventDefault();
  const title = document.getElementById('theory-title').value.trim();
  const text = document.getElementById('theory-text').value.trim();
  const gameId = document.getElementById('theory-game').value;
  const game = GAMES[gameId];
  const card = document.createElement('article');
  card.className = 'glass rounded-2xl p-6 post';
  card.innerHTML = `
    <h2 class="font-cinzel text-xl mb-2">${title}</h2>
    <p class="text-white/60 text-sm">${text}</p>
    <div class="mt-4 flex flex-wrap items-center gap-3 text-xs">
      <button class="like-btn flex items-center gap-1 text-white/50 hover:text-red-400 transition" data-likes="0">♡ <span>0</span></button>
      <button class="comment-btn text-white/50 hover:text-nexus transition">💬 Comment</button>
      <span class="tag rounded-full px-3 py-1">#${game.title}</span><span class="tag rounded-full px-3 py-1">#Theory</span>
    </div>`;
  document.getElementById('lore-feed').prepend(card);
  e.target.reset();
  closeModal(document.getElementById('theory-modal'));
});
document.addEventListener('click', e => {
  const like = e.target.closest('.like-btn');
  if (like) {
    const count = like.querySelector('span');
    let n = parseInt(like.dataset.likes || count.textContent, 10) + 1;
    like.dataset.likes = n;
    count.textContent = n;
    like.classList.add('text-red-400');
  }
  const comment = e.target.closest('.comment-btn');
  if (comment) showToast('Comments coming soon');
});
function showToast(msg) {
  const t = document.createElement('div');
  t.className = 'fixed bottom-6 left-1/2 -translate-x-1/2 glass rounded-full px-5 py-2 text-sm z-[90]';
  t.textContent = msg;
  document.body.appendChild(t);
  gsap.fromTo(t, {opacity:0, y:20}, {opacity:1, y:0, duration:.3});
  setTimeout(() => gsap.to(t, {opacity:0, y:20, duration:.3, onComplete: () => t.remove()}), 1800);
}
</script>

<script>
const pageTransition = document.getElementById('page-transition');
document.querySelectorAll('a[href$=".html"]').forEach(a => {
  a.addEventListener('click', e => {
    e.preventDefault();
    gsap.to(pageTransition, { opacity:1, duration:.35, onComplete: () => { location.href = a.href; } });
  });
});
function openModal(el){
  el.style.display='flex';
  gsap.fromTo(el, {opacity:0}, {opacity:1, duration:.25});
  gsap.fromTo(el.querySelector('.modal-card'), {scale:.92, opacity:0}, {scale:1, opacity:1, duration:.35, ease:'back.out(1.4)'});
}
function closeModal(el){
  gsap.to(el, {opacity:0, duration:.25, onComplete: () => { el.style.display='none'; } });
}
</script>
