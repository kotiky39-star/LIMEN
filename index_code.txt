function spawnParticles(el, n, color, mode = 'rise') {
  el.innerHTML = '';
  for (let i = 0; i < n; i++) {
    const s = document.createElement('span');
    s.className = `ptc ${mode}`;
    let size, dur, pos;
    if (mode === 'fall') { size = Math.random() * 3 + 1.5; dur = 6 + Math.random() * 6; pos = `left:${Math.random() * 100}%;`; }
    else if (mode === 'drift') { size = Math.random() * 2 + 1; dur = 2.5 + Math.random() * 2.5; pos = `top:${Math.random() * 100}%;`; }
    else { size = Math.random() * 3 + 1; dur = 8 + Math.random() * 10; pos = `left:${Math.random() * 100}%;`; }
    s.style.cssText = `${pos}width:${size}px;height:${size}px;animation-duration:${dur}s;animation-delay:${-Math.random() * dur}s;` +
      `--dx:${(Math.random() - 0.5) * 220}px;--dy:${(Math.random() - 0.5) * 160}px;background:${color};box-shadow:0 0 ${size * 4}px ${color};`;
    el.appendChild(s);
  }
}

}

  for (let i = 0; i < n; i++) {

    const f = { x: (Math.random() - 0.5) * 2 * amp * k, skewX: (Math.random() - 0.5) * 6 * k, duration: d, ease: 'none' };
    if (withY) f.y = (Math.random() - 0.5) * 1.2 * amp * k;
    frames.push(f);
  }
  const end = { x: 0, skewX: 0, duration: d, ease: 'none' };
  if (withY) end.y = 0;
  frames.push(end);
  return frames;
}

function makeSprite(r, g, b) {
  const s = document.createElement('canvas'); s.width = s.height = 64;
  const c = s.getContext('2d');
  const gr = c.createRadialGradient(32, 32, 0, 32, 32, 32);
  gr.addColorStop(0, 'rgba(255,255,255,1)');
  gr.addColorStop(0.22, `rgba(${r},${g},${b},0.9)`);
  gr.addColorStop(1, `rgba(${r},${g},${b},0)`);
  c.fillStyle = gr; c.fillRect(0, 0, 64, 64);
  return s;
}
const SPR = { gold: makeSprite(255, 205, 110), fire: makeSprite(255, 130, 40) };

function sigilPoint(u) {
  const a = Math.random() * Math.PI * 2;
  if (u < 0.38) return [Math.cos(a), Math.sin(a)];
  if (u < 0.55) return [0, -1.25 + Math.random() * 2.6];
  if (u < 0.75) return [Math.cos(a) * 0.62, -0.38 + Math.sin(a) * 0.62];
  if (u < 0.9) return [Math.cos(a) * 0.42, 0.42 + Math.sin(a) * 0.42];
  return [Math.cos(a) * 0.2, -0.05 + Math.sin(a) * 0.2];
}
