  {title:'Legendary Animals', img:'https://picsum.photos/seed/legend/800/400', text:'Myths of the frontier. Each hunt is a test of patience, skill, and respect for the wild.'},

];
document.querySelectorAll('.codex-card').forEach((card, i) => {
  card.style.cursor='pointer';
  card.addEventListener('click', () => {
    const d = CODEX[i % CODEX.length];
    document.getElementById('codex-title').textContent = d.title;
    const read = JSON.parse(localStorage.getItem('readArticles') || '[]');
    read.push(gameId);
    localStorage.setItem('readArticles', JSON.stringify(read));
    document.getElementById('codex-text').innerHTML = d.text;
    document.getElementById('codex-img').src = d.img;
    openModal(document.getElementById('codex-modal'));
  });
});
document.getElementById('codex-close').addEventListener('click', () => closeModal(document.getElementById('codex-modal')));
const NESTED = {
  pinkertons: { title:'Pinkertons', text:'A ruthless detective agency hired to hunt the gang across states.' },
  vanderlinde: { title:'Van der Linde Gang', text:'A found family that slowly became a cult of personality.' }
};
document.getElementById('codex-text').addEventListener('click', e => {
  const link = e.target.closest('.lore-link');
  if (link) {
    const d = NESTED[link.dataset.target];
    if (!d) return;
    document.getElementById('lore-modal-title').textContent = d.title;
    document.getElementById('lore-modal-text').textContent = d.text;
    openModal(document.getElementById('lore-modal'));
  }
});
document.getElementById('lore-close').addEventListener('click', () => closeModal(document.getElementById('lore-modal')));
</script>

<script>
const pageTransition = document.getElementById('page-transition');
document.querySelectorAll('a[href$=".html"]').forEach(a => {
  a.addEventListener('click', e => {
    e.preventDefault();
    gsap.to(pageTransition, { opacity:1, duration:.35, onComplete: () => { location.href = a.href; } });
  });
});
function openModal(el){
  el.style.display='flex';
  gsap.fromTo(el, {opacity:0}, {opacity:1, duration:.25});
  gsap.fromTo(el.querySelector('.modal-card'), {scale:.92, opacity:0}, {scale:1, opacity:1, duration:.35, ease:'back.out(1.4)'});
}
function closeModal(el){
  gsap.to(el, {opacity:0, duration:.25, onComplete: () => { el.style.display='none'; } });
}
</script>
