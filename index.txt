<script>
  const lenis = new Lenis({duration:1.1});
  function raf(t){ lenis.raf(t); requestAnimationFrame(raf); }
  requestAnimationFrame(raf);

  const map = document.getElementById('map');
  window.addEventListener('mousemove', e => {
    const x = (e.clientX/window.innerWidth - .5)*40;
    const y = (e.clientY/window.innerHeight - .5)*40;
    gsap.to(map, {x,y,duration:1,ease:'power2.out'});
  });

  gsap.from('.node', {opacity:0, scale:0, stagger:.12, duration:.8, ease:'back.out(1.6)'});
  gsap.from('.line', {opacity:0, stagger:.1, duration:1.2});

  document.querySelectorAll('.node').forEach(n=>{
    n.addEventListener('click',()=>{
      gsap.timeline()
        .to(n,{scale:3, boxShadow:'0 0 90px #38bdf8', duration:.3})
        .to(map,{opacity:0, scale:1.3, duration:.5},0)
        .to(pageTransition,{opacity:1, duration:.5},0)
        .add(()=>{ window.location.href = 'universe.html?game=' + n.dataset.game; },0.8);
    });
  });
</script>

<script>
const pageTransition = document.getElementById('page-transition');
document.querySelectorAll('a[href$=".html"]').forEach(a => {
  a.addEventListener('click', e => {
    e.preventDefault();
    gsap.to(pageTransition, { opacity:1, duration:.35, onComplete: () => { location.href = a.href; } });
  });
});
function openModal(el){
  el.style.display='flex';
  gsap.fromTo(el, {opacity:0}, {opacity:1, duration:.25});
  gsap.fromTo(el.querySelector('.modal-card'), {scale:.92, opacity:0}, {scale:1, opacity:1, duration:.35, ease:'back.out(1.4)'});
}
function closeModal(el){
  gsap.to(el, {opacity:0, duration:.25, onComplete: () => { el.style.display='none'; } });
}
</script>
