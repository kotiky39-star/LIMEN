<script>
  const lenis = new Lenis({duration:1.1});
  function raf(t){ lenis.raf(t); requestAnimationFrame(raf); }
  requestAnimationFrame(raf);
  gsap.from('.badge', {opacity:0, y:30, stagger:.08, duration:.7});
  const COLLECTIONS_KEY = 'nexusCollections';
  function renderCollections() {
    const grid = document.getElementById('collections-grid');
    grid.innerHTML = '';
    JSON.parse(localStorage.getItem(COLLECTIONS_KEY) || '[]').forEach(c => {
      const card = document.createElement('div');
      card.className = 'glass rounded-2xl p-5';
      card.innerHTML = `<h3 class="font-cinzel text-lg">${c.title}</h3><p class="text-white/50 text-sm">${(GAMES[c.game] || {}).title || c.game}</p>`;
      grid.appendChild(card);
    });
  }
  const collectionSelect = document.getElementById('collection-game');
  Object.values(GAMES).forEach(g => {
    const o = document.createElement('option');
    o.value = g.id; o.textContent = g.title;
    collectionSelect.appendChild(o);
  });
  document.getElementById('create-collection-btn').addEventListener('click', () => openModal(document.getElementById('collection-modal')));
  document.getElementById('collection-form').addEventListener('submit', e => {
    e.preventDefault();
    const list = JSON.parse(localStorage.getItem(COLLECTIONS_KEY) || '[]');
    list.push({ title: document.getElementById('collection-title').value.trim(), game: document.getElementById('collection-game').value });
    localStorage.setItem(COLLECTIONS_KEY, JSON.stringify(list));
    renderCollections();
    e.target.reset();
    closeModal(document.getElementById('collection-modal'));
  });
  renderCollections();
  function refreshGamification() {
    const read = JSON.parse(localStorage.getItem('readArticles') || '[]');
    const counts = read.reduce((a, g) => (a[g] = (a[g] || 0) + 1, a), {});
    let unlocked = 0;
    document.querySelectorAll('.badge').forEach(badge => {
      const isUnlocked = (counts[badge.dataset.game] || 0) >= 3;
      badge.classList.toggle('locked', !isUnlocked);
      badge.classList.toggle('unlocked', isUnlocked);
      if (isUnlocked) unlocked++;
    });
    document.getElementById('xp-fill').style.width = (unlocked / 5 * 100) + '%';
    document.getElementById('xp-text').textContent = (1200 + unlocked * 400) + ' / 3200 XP';
  }
  refreshGamification();
  document.querySelectorAll('.tab-btn').forEach(btn => {
    btn.addEventListener('click', () => {
      document.querySelectorAll('.tab-btn').forEach(b => b.classList.remove('is-active'));
      btn.classList.add('is-active');
      document.querySelectorAll('.tab-panel').forEach(p => {
        if (p.id === `tab-${btn.dataset.tab}`) {
          p.classList.remove('hidden');
          gsap.fromTo(p, {opacity:0, y:12}, {opacity:1, y:0, duration:.3});
        } else p.classList.add('hidden');
      });
    });
  });
  document.getElementById('settings-form').addEventListener('submit', e => {
    e.preventDefault();
    const name = document.getElementById('username-input').value.trim();
    const color = document.getElementById('avatar-color').value;
    if (name) document.getElementById('sidebar-name').textContent = name;
    document.getElementById('sidebar-avatar').className = `w-20 h-20 rounded-full bg-gradient-to-br ${color} grid place-items-center text-2xl font-semibold mx-auto`;
  });
</script>

<script>
const pageTransition = document.getElementById('page-transition');
document.querySelectorAll('a[href$=".html"]').forEach(a => {
  a.addEventListener('click', e => {
    e.preventDefault();
    gsap.to(pageTransition, { opacity:1, duration:.35, onComplete: () => { location.href = a.href; } });
  });
});
function openModal(el){
  el.style.display='flex';
  gsap.fromTo(el, {opacity:0}, {opacity:1, duration:.25});
  gsap.fromTo(el.querySelector('.modal-card'), {scale:.92, opacity:0}, {scale:1, opacity:1, duration:.35, ease:'back.out(1.4)'});
}
function closeModal(el){
  gsap.to(el, {opacity:0, duration:.25, onComplete: () => { el.style.display='none'; } });
}
</script>
